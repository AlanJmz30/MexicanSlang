
import { useState } from 'react';
import { Pressable, ScrollView, StyleSheet, Text, View } from 'react-native';

const expressions = [
  {
    phrase: '¡Qué padre!',
    meaning: 'That’s awesome!',
    example: '¡Qué padre está tu departamento!',
    emoji: '😎',
  },
  {
    phrase: 'No manches',
    meaning: 'No way! / You’re kidding!',
    example: '¿Ganaste un viaje? ¡No manches!',
    emoji: '🤯',
  },
  {
    phrase: '¡Aguas!',
    meaning: 'Watch out!',
    example: '¡Aguas! Viene un coche.',
    emoji: '⚠️',
  },
];

export default function Index() {
  const [answer, setAnswer] = useState('');
  const [score, setScore] = useState(0);
  const correctAnswer = 'Watch out!';
  return (

    <ScrollView style={styles.container}>
      <View style={styles.header}>
        <Text style={styles.flag}>🇲🇽</Text>
        <Text style={styles.title}>MexicanSlang</Text>
        <Text style={styles.subtitle}>
          Speak Mexican, not just Spanish.
        </Text>
      </View>

      <View style={styles.banner}>
        <Text style={styles.bannerTitle}>¡Qué onda! 👋</Text>
        <Text style={styles.bannerText}>
          Learn the real Spanish people speak in Mexico.
        </Text>
      </View>

      <Text style={styles.sectionTitle}>
        Your first 3 expressions
      </Text>

      {expressions.map((item) => (
        <View key={item.phrase} style={styles.card}>
          <Text style={styles.emoji}>{item.emoji}</Text>
          <Text style={styles.phrase}>{item.phrase}</Text>
          <Text style={styles.meaning}>{item.meaning}</Text>
          <Text style={styles.example}>{item.example}</Text>
        </View>
      ))}


      <View style={styles.quiz}>
        <Text style={styles.quizTitle}>Quick challenge 🎯</Text>
        <Text style={styles.score}>Score: {score} ⭐</Text>
        <Text style={styles.quizQuestion}>
          What does “¡Aguas!” mean?
        </Text>

        {['Good luck!', 'Watch out!', 'Let’s go!'].map((option) => (
          <Pressable
            key={option}
            style={[
              styles.option,
              answer === option && {
                backgroundColor:
                  option === correctAnswer ? '#DCFCE7' : '#FEE2E2',
              },
            ]}

            onPress={() => {
              if (answer !== '') return;

              if (option === correctAnswer) {
                setScore((previous) => previous + 1);
              }

              setAnswer(option);
            }}
          >
            <Text style={styles.optionText}>{option}</Text>
          </Pressable>
        ))}

        {answer !== '' && (
          <Text style={styles.feedback}>
            {answer === correctAnswer
              ? 'Correct! 🎉 ¡Aguas! means “Watch out!”'
              : 'Not quite! Try again.'}
          </Text>
        )}
      </View>


      <Text style={styles.footer}>
        Made with ❤️ in Mexico
      </Text>
    </ScrollView>
  );
}
